\section{Angular Momentum Commutation Algebra}

\subsection{Fundamental Commutators}

To establish the Lie algebra of the rotation group $SO(3)$, we must compute the commutation relations between the components of the angular momentum operator explicitly. We start from the canonical commutation relations for position and momentum:

\begin{equation}
    [\hat{x}_i, \hat{p}_j] = i\hbar\delta_{ij}, \quad [\hat{x}_i, \hat{x}_j] = 0, \quad [\hat{p}_i, \hat{p}_j] = 0.
\end{equation}

Using the explicit definitions derived in the previous section:
\begin{equation}
    \hat{L}_x = \hat{y}\hat{p}_z - \hat{z}\hat{p}_y, \quad \hat{L}_y = \hat{z}\hat{p}_x - \hat{x}\hat{p}_z, \quad \hat{L}_z = \hat{x}\hat{p}_y - \hat{y}\hat{p}_x.
\end{equation}

\subsubsection{Commutator $[\hat{L}_x, \hat{L}_y]$}

We substitute the explicit expressions for the operators:

\begin{equation}
    [\hat{L}_x, \hat{L}_y] = [(\hat{y}\hat{p}_z - \hat{z}\hat{p}_y), (\hat{z}\hat{p}_x - \hat{x}\hat{p}_z)]
\end{equation}

Using the linearity of the commutator $[A+B, C+D] = [A,C] + [A,D] + [B,C] + [B,D]$, we expand this expression into four terms:

\begin{equation}
    [\hat{L}_x, \hat{L}_y] = \underbrace{[\hat{y}\hat{p}_z, \hat{z}\hat{p}_x]}_{\text{I}} - \underbrace{[\hat{y}\hat{p}_z, \hat{x}\hat{p}_z]}_{\text{II}} - \underbrace{[\hat{z}\hat{p}_y, \hat{z}\hat{p}_x]}_{\text{III}} + \underbrace{[\hat{z}\hat{p}_y, \hat{x}\hat{p}_z]}_{\text{IV}}
\end{equation}

We analyze each term separately using the product rule $[AB, CD] = A[B,C]D + AC[B,D] + [A,C]BD + C[A,D]B$ and the fact that operators acting on different Cartesian coordinates commute (e.g., $[\hat{y}, \hat{p}_x] = 0$).

\begin{itemize}
    \item \textbf{Term I:} $[\hat{y}\hat{p}_z, \hat{z}\hat{p}_x]$. Here, $\hat{y}$ and $\hat{p}_x$ commute with everything in the bracket. The non-commuting pair is $\hat{p}_z$ and $\hat{z}$.
    \begin{equation}
        [\hat{y}\hat{p}_z, \hat{z}\hat{p}_x] = \hat{y}\hat{p}_x [\hat{p}_z, \hat{z}] = \hat{y}\hat{p}_x (-i\hbar) = -i\hbar \hat{y}\hat{p}_x
    \end{equation}
    (Recall that $[\hat{z}, \hat{p}_z] = i\hbar$, so $[\hat{p}_z, \hat{z}] = -i\hbar$).

    \item \textbf{Term II:} $[\hat{y}\hat{p}_z, \hat{x}\hat{p}_z]$. All operators involve different coordinates ($y, z, x$) or the same operator ($\hat{p}_z$ with $\hat{p}_z$), so they all commute.
    \begin{equation}
        [\hat{y}\hat{p}_z, \hat{x}\hat{p}_z] = 0
    \end{equation}

    \item \textbf{Term III:} $[\hat{z}\hat{p}_y, \hat{z}\hat{p}_x]$. Similarly, $\hat{z}$ commutes with $\hat{z}$, and $\hat{p}_y$ commutes with $\hat{p}_x$.
    \begin{equation}
        [\hat{z}\hat{p}_y, \hat{z}\hat{p}_x] = 0
    \end{equation}

    \item \textbf{Term IV:} $[\hat{z}\hat{p}_y, \hat{x}\hat{p}_z]$. Here, $\hat{p}_y$ and $\hat{x}$ commute with the others. The non-commuting pair is $\hat{z}$ and $\hat{p}_z$.
    \begin{equation}
        [\hat{z}\hat{p}_y, \hat{x}\hat{p}_z] = \hat{x}\hat{p}_y [\hat{z}, \hat{p}_z] = \hat{x}\hat{p}_y (i\hbar) = i\hbar \hat{x}\hat{p}_y
    \end{equation}
\end{itemize}

Substituting these results back into the expansion:

\begin{equation}
    [\hat{L}_x, \hat{L}_y] = (-i\hbar \hat{y}\hat{p}_x) - 0 - 0 + (i\hbar \hat{x}\hat{p}_y)
\end{equation}
\begin{equation}
    [\hat{L}_x, \hat{L}_y] = i\hbar (\hat{x}\hat{p}_y - \hat{y}\hat{p}_x)
\end{equation}

Recognizing the term in parentheses as $\hat{L}_z$, we arrive at the first fundamental relation:
\begin{equation}
    [\hat{L}_x, \hat{L}_y] = i\hbar \hat{L}_z
\end{equation}

\subsubsection{Commutator $[\hat{L}_y, \hat{L}_z]$}

Substituting the explicit forms:
\begin{equation}
    [\hat{L}_y, \hat{L}_z] = [(\hat{z}\hat{p}_x - \hat{x}\hat{p}_z), (\hat{x}\hat{p}_y - \hat{y}\hat{p}_x)]
\end{equation}

Expanding the terms similarly:
\begin{align}
    [\hat{z}\hat{p}_x, \hat{x}\hat{p}_y] &= \hat{z}\hat{p}_y [\hat{p}_x, \hat{x}] = \hat{z}\hat{p}_y (-i\hbar) \\
    [\hat{z}\hat{p}_x, \hat{y}\hat{p}_x] &= 0 \\
    [\hat{x}\hat{p}_z, \hat{x}\hat{p}_y] &= 0 \\
    [\hat{x}\hat{p}_z, \hat{y}\hat{p}_x] &= \hat{y}\hat{p}_z [\hat{x}, \hat{p}_x] = \hat{y}\hat{p}_z (i\hbar)
\end{align}

Reassembling (paying attention to the negative signs in the original expansion):
\begin{equation}
    [\hat{L}_y, \hat{L}_z] = (-i\hbar \hat{z}\hat{p}_y) - (0) - (0) + (-1)(-1)(i\hbar \hat{y}\hat{p}_z)
\end{equation}
\begin{equation}
    [\hat{L}_y, \hat{L}_z] = i\hbar (\hat{y}\hat{p}_z - \hat{z}\hat{p}_y) = i\hbar \hat{L}_x
\end{equation}

\subsubsection{Commutator $[\hat{L}_z, \hat{L}_x]$}

Substituting the explicit forms:
\begin{equation}
    [\hat{L}_z, \hat{L}_x] = [(\hat{x}\hat{p}_y - \hat{y}\hat{p}_x), (\hat{y}\hat{p}_z - \hat{z}\hat{p}_y)]
\end{equation}

Expanding the terms:
\begin{align}
    [\hat{x}\hat{p}_y, \hat{y}\hat{p}_z] &= \hat{x}\hat{p}_z [\hat{p}_y, \hat{y}] = \hat{x}\hat{p}_z (-i\hbar) \\
    [\hat{x}\hat{p}_y, \hat{z}\hat{p}_y] &= 0 \\
    [\hat{y}\hat{p}_x, \hat{y}\hat{p}_z] &= 0 \\
    [\hat{y}\hat{p}_x, \hat{z}\hat{p}_y] &= \hat{z}\hat{p}_x [\hat{y}, \hat{p}_y] = \hat{z}\hat{p}_x (i\hbar)
\end{align}

Reassembling:
\begin{equation}
    [\hat{L}_z, \hat{L}_x] = (-i\hbar \hat{x}\hat{p}_z) - 0 - 0 - (-1)(i\hbar \hat{z}\hat{p}_x)
\end{equation}
\begin{equation}
    [\hat{L}_z, \hat{L}_x] = i\hbar (\hat{z}\hat{p}_x - \hat{x}\hat{p}_z) = i\hbar \hat{L}_y
\end{equation}

These three results can be summarized compactly using the Levi-Civita antisymmetric tensor, confirming that the angular momentum components satisfy the Lie algebra of $SO(3)$:

\begin{equation}
    [\hat{L}_i, \hat{L}_j] = i\hbar \epsilon_{ijk} \hat{L}_k
\end{equation}

\subsection{The Operator $\hat{L}^2$ and its Commutators}

\subsubsection{Definition}
We define the operator corresponding to the square of the magnitude of the angular momentum vector as the sum of the squares of its Cartesian components:

\begin{equation}
    \hat{L}^2 = \hat{L}_x^2 + \hat{L}_y^2 + \hat{L}_z^2
\end{equation}

\subsubsection{Commutation with Components}
We now wish to demonstrate that the scalar operator $\hat{L}^2$ commutes with any specific component of the vector operator $\hat{L}$. Without loss of generality, we choose to calculate the commutator with $\hat{L}_z$. Due to the rotational symmetry of the indices, the result will be valid for $\hat{L}_x$ and $\hat{L}_y$ as well.

\begin{equation}
    [\hat{L}^2, \hat{L}_z] = [\hat{L}_x^2 + \hat{L}_y^2 + \hat{L}_z^2, \hat{L}_z]
\end{equation}

Using the linearity of the commutator, we separate this into three terms:
\begin{equation}
    [\hat{L}^2, \hat{L}_z] = [\hat{L}_x^2, \hat{L}_z] + [\hat{L}_y^2, \hat{L}_z] + [\hat{L}_z^2, \hat{L}_z]
\end{equation}

The last term vanishes trivially because any operator commutes with itself and powers of itself ($[\hat{A}^n, \hat{A}] = 0$):
\begin{equation}
    [\hat{L}_z^2, \hat{L}_z] = 0
\end{equation}

For the remaining terms, we use the property $[A^2, B] = A[A,B] + [A,B]A$.

\textbf{1. Calculating $[\hat{L}_x^2, \hat{L}_z]$:}
Using the fundamental relation $[\hat{L}_x, \hat{L}_z] = -i\hbar \hat{L}_y$ (derived from $[\hat{L}_z, \hat{L}_x] = i\hbar \hat{L}_y$):
\begin{align}
    [\hat{L}_x^2, \hat{L}_z] &= \hat{L}_x [\hat{L}_x, \hat{L}_z] + [\hat{L}_x, \hat{L}_z] \hat{L}_x \nonumber \\
    &= \hat{L}_x (-i\hbar \hat{L}_y) + (-i\hbar \hat{L}_y) \hat{L}_x \nonumber \\
    &= -i\hbar (\hat{L}_x \hat{L}_y + \hat{L}_y \hat{L}_x)
\end{align}

\textbf{2. Calculating $[\hat{L}_y^2, \hat{L}_z]$:}
Using the fundamental relation $[\hat{L}_y, \hat{L}_z] = i\hbar \hat{L}_x$:
\begin{align}
    [\hat{L}_y^2, \hat{L}_z] &= \hat{L}_y [\hat{L}_y, \hat{L}_z] + [\hat{L}_y, \hat{L}_z] \hat{L}_y \nonumber \\
    &= \hat{L}_y (i\hbar \hat{L}_x) + (i\hbar \hat{L}_x) \hat{L}_y \nonumber \\
    &= i\hbar (\hat{L}_y \hat{L}_x + \hat{L}_x \hat{L}_y)
\end{align}

\textbf{Total Commutator:}
Adding the results:
\begin{equation}
    [\hat{L}^2, \hat{L}_z] = -i\hbar (\hat{L}_x \hat{L}_y + \hat{L}_y \hat{L}_x) + i\hbar (\hat{L}_y \hat{L}_x + \hat{L}_x \hat{L}_y)
\end{equation}
Since scalar multiplication is distributive, the terms cancel exactly:
\begin{equation}
    [\hat{L}^2, \hat{L}_z] = 0
\end{equation}

By symmetry, it follows that:
\begin{equation}
    [\hat{L}^2, \hat{L}_i] = 0, \quad \text{for } i = x, y, z
\end{equation}

\subsubsection{Physical Significance: Simultaneous Eigenstates}
The fact that $[\hat{L}^2, \hat{L}_z] = 0$ has profound physical implications in quantum mechanics. Two commuting Hermitian operators share a complete set of simultaneous eigenstates.

This means we can specify the state of the system using the eigenvalues of both operators simultaneously without violating the Heisenberg Uncertainty Principle. Physically, this implies that the magnitude of the angular momentum ($\hat{L}^2$) and its projection along one axis (arbitrarily chosen as the z-axis, $\hat{L}_z$) are compatible observables.

However, since different components do not commute (e.g., $[\hat{L}_x, \hat{L}_y] \neq 0$), we cannot determine all three components simultaneously. Thus, the complete description of the angular momentum state is given by the basis kets $|l, m\rangle$, characterized by the quantum numbers $l$ and $m$:
\begin{itemize}
    \item $l$ relates to the eigenvalue of $\hat{L}^2$ ($\hbar^2 l(l+1)$).
    \item $m$ relates to the eigenvalue of $\hat{L}_z$ ($\hbar m$).
\end{itemize}
