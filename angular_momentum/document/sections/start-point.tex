\section{Starting Point: Known Results}

In this section, we summarize the fundamental elements of the quantum mechanical formalism for a free particle of mass $m$. These results are assumed to be known and serve as the foundation for constructing the angular momentum representation. Proofs are omitted; only the necessary definitions and operational relations are presented.

\subsection{Free Particle Solution in Different Representations}

The free particle Hamiltonian is given by $\hat{H} = \frac{\hat{\mathbf{p}}^2}{2m}$. The wavefunctions in the various representations evolve as follows:

\begin{itemize}
    \item \textbf{Position Representation ($\psi(\mathbf{r}, t)$):}
    Satisfies the time-dependent Schrödinger equation:
    \begin{equation}
        i\hbar \frac{\partial}{\partial t}\psi(\mathbf{r},t) = -\frac{\hbar^2}{2m}\nabla^2\psi(\mathbf{r},t).
    \end{equation}
    
    \item \textbf{Momentum Representation ($\psi(\mathbf{p}, t)$):}
    Since $\hat{\mathbf{p}}$ commutes with $\hat{H}$ for the free particle, the evolution is a multiplicative phase:
    \begin{equation}
        \psi(\mathbf{p},t) = \psi(\mathbf{p},0) \exp\left(-\frac{i}{\hbar}\frac{p^2}{2m}t\right).
    \end{equation}
    
    \item \textbf{Energy Representation ($\psi(E, t)$):}
    For an energy eigenstate with $E = p^2/2m$:
    \begin{equation}
        \psi(E,t) = \psi(E,0) \exp\left(-\frac{i}{\hbar}Et\right).
    \end{equation}
\end{itemize}

\subsection{Integral Transformations Between Representations}

The connection between different bases is established through integral transformations defined by the inner products (kernels) of the basis states.

\subsubsection{Fourier Transform ($\mathbf{r} \leftrightarrow \mathbf{p}$)}
The wavefunctions are related by:
\begin{align}
    \psi(\mathbf{r},t) &= \frac{1}{(2\pi\hbar)^{3/2}} \int d^3p \, e^{i\mathbf{p}\cdot\mathbf{r}/\hbar} \psi(\mathbf{p},t), \\
    \psi(\mathbf{p},t) &= \frac{1}{(2\pi\hbar)^{3/2}} \int d^3r \, e^{-i\mathbf{p}\cdot\mathbf{r}/\hbar} \psi(\mathbf{r},t).
\end{align}

\subsubsection{Momentum-Energy Transformation ($\mathbf{p} \leftrightarrow E$)}
Using the dispersion relation $E = p^2/2m$ and the properties of the deformed Dirac delta:
\begin{equation}
    \delta\left(E - \frac{p^2}{2m}\right) = \frac{m}{p}\delta(p - \sqrt{2mE}).
\end{equation}
This allows expressing integrals over momentum as integrals over energy, facilitating the transition from $\psi(\mathbf{p})$ to spectral energy densities.

\subsection{Eigenvalue Equations and Basic Operators}

We define the fundamental operators and their eigenstates, which form complete and orthonormal bases.

\begin{itemize}
    \item \textbf{Position Operator $\hat{\mathbf{r}}$:} 
    \begin{equation}
        \hat{\mathbf{r}}|\mathbf{r}\rangle = \mathbf{r}|\mathbf{r}\rangle, \quad \int d^3r |\mathbf{r}\rangle\langle\mathbf{r}| = \hat{\mathbb{I}}.
    \end{equation}
    
    \item \textbf{Momentum Operator $\hat{\mathbf{p}}$:}
    \begin{equation}
        \hat{\mathbf{p}}|\mathbf{p}\rangle = \mathbf{p}|\mathbf{p}\rangle, \quad \int d^3p |\mathbf{p}\rangle\langle\mathbf{p}| = \hat{\mathbb{I}}.
    \end{equation}
    In the position basis, the momentum operator acts as the generator of translations: $\hat{\mathbf{p}} = -i\hbar\nabla_\mathbf{r}$.
    
    \item \textbf{Free Hamiltonian $\hat{H}$:}
    Its eigenstates $|E\rangle$ coincide (with degeneracy) with the momentum states:
    \begin{equation}
        \hat{H}|\mathbf{p}\rangle = \frac{p^2}{2m}|\mathbf{p}\rangle.
    \end{equation}
    
    \item \textbf{Inner Products (Transformation Kernels):}
    \begin{equation}
        \langle \mathbf{r} | \mathbf{p} \rangle = \frac{1}{(2\pi\hbar)^{3/2}}e^{i\mathbf{p}\cdot\mathbf{r}/\hbar}, \quad \langle \mathbf{p} | E \rangle \propto \delta(E - p^2/2m).
    \end{equation}
\end{itemize}

\subsection{Symmetry Operators and Ehrenfest's Theorem}

The following unitary operators generate the continuous transformations relevant to the system.

\subsubsection{Translation and Evolution Operators}

\begin{itemize}
    \item \textbf{Spatial Translation:} Generated by the momentum operator $\hat{\mathbf{p}}$. For a finite vector $\mathbf{a}$ and an infinitesimal one $d\mathbf{a}$:
    \begin{equation}
        \hat{T}(\mathbf{a}) = \exp\left(-\frac{i}{\hbar}\mathbf{a}\cdot\hat{\mathbf{p}}\right), \quad \hat{T}(d\mathbf{a}) \approx \hat{\mathbb{I}} - \frac{i}{\hbar}d\mathbf{a}\cdot\hat{\mathbf{p}}.
    \end{equation}
    
    \item \textbf{Time Evolution:} Generated by the Hamiltonian $\hat{H}$.
    \begin{equation}
        \hat{U}(t) = \exp\left(-\frac{i}{\hbar}\hat{H}t\right), \quad \hat{U}(dt) \approx \hat{\mathbb{I}} - \frac{i}{\hbar}\hat{H}dt.
    \end{equation}
    
    \item \textbf{Momentum Space Translation:} Analogous to spatial translation, the position operator $\hat{\mathbf{r}}$ generates displacements in momentum space (boosts). For a momentum displacement $\mathbf{q}$:
    \begin{equation}
        \hat{T}_{\mathbf{p}}(\mathbf{q}) = \exp\left(\frac{i}{\hbar}\mathbf{q}\cdot\hat{\mathbf{r}}\right).
    \end{equation}
    The action of these operators defines the change of basis between the Hilbert spaces associated with the observables.
\end{itemize}

\subsubsection{Correspondence Principle and Ehrenfest's Theorem}

The connection to classical mechanics is formally established via Ehrenfest's Theorem, which governs the time evolution of expectation values. For an operator $\hat{A}$ with no explicit time dependence:
\begin{equation}
    \frac{d}{dt}\langle \hat{A} \rangle = \frac{1}{i\hbar}\langle [\hat{A}, \hat{H}] \rangle.
\end{equation}

Applied to the free particle ($\hat{H} = \hat{p}^2/2m$):
\begin{itemize}
    \item \textbf{Position:} $\frac{d}{dt}\langle \hat{\mathbf{r}} \rangle = \frac{1}{m}\langle \hat{\mathbf{p}} \rangle$ (Velocity-Momentum relation).
    \item \textbf{Momentum:} $\frac{d}{dt}\langle \hat{\mathbf{p}} \rangle = 0$ (Conservation of linear momentum due to translational invariance).
\end{itemize}
These results ensure that wave packets follow classical trajectories in the macroscopic limit, satisfying the correspondence principle.
